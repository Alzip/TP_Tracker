cmake_minimum_required(VERSION 3.26)
project(TPTracker LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(TPTRACKER_TESTS_ONLY "Build portable tracker tests without the game SDK" OFF)
set(DUSKLIGHT_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/.deps/dusklight" CACHE PATH "External Dusklight source checkout")
if(NOT EXISTS "${DUSKLIGHT_SOURCE_DIR}/sdk/include/mods/svc/ui.h")
    message(FATAL_ERROR "Run python tools/bootstrap.py or set -DDUSKLIGHT_SOURCE_DIR=/path/to/dusklight")
endif()
include(FetchContent)
# Headers only: input comes from the host's SDL events, never a second SDL instance.
FetchContent_Declare(tptracker_sdl
    URL https://github.com/libsdl-org/SDL/archive/refs/tags/release-3.4.10.tar.gz
    SOURCE_SUBDIR tptracker-headers-only)
FetchContent_MakeAvailable(tptracker_sdl)
if(NOT TPTRACKER_TESTS_ONLY)
    if(NOT TARGET dusklight_mod_api)
        add_subdirectory("${DUSKLIGHT_SOURCE_DIR}/sdk" dusklight-sdk EXCLUDE_FROM_ALL)
    endif()
    add_mod(TP_Randomizer_Tracker FEATURES game fmt
        SOURCES src/mod.cpp src/model.cpp src/seed.cpp src/map.cpp src/controller.cpp MOD_JSON mod.json RES_DIR res)
    target_include_directories(TP_Randomizer_Tracker PRIVATE "${DUSKLIGHT_SOURCE_DIR}/extern/aurora/include")
    target_include_directories(TP_Randomizer_Tracker PRIVATE "${tptracker_sdl_SOURCE_DIR}/include")
    if(APPLE AND CMAKE_SYSTEM_PROCESSOR STREQUAL "x86_64")
        # The SDK's Intel stub omits LC_UUID, which the new Apple linker rejects.
        target_link_options(TP_Randomizer_Tracker PRIVATE "LINKER:-ld_classic")
    endif()
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(tptracker_input_tests tests/input_tests.cpp src/controller.cpp)
    target_compile_definitions(tptracker_input_tests PRIVATE TPTRACKER_INPUT_TESTS)
    target_include_directories(tptracker_input_tests PRIVATE src "${tptracker_sdl_SOURCE_DIR}/include" "${DUSKLIGHT_SOURCE_DIR}/sdk/include")
    add_test(NAME tptracker_input COMMAND tptracker_input_tests)
    add_executable(tptracker_tests tests/model_tests.cpp src/model.cpp)
    target_include_directories(tptracker_tests PRIVATE src)
    add_test(NAME tptracker_model COMMAND tptracker_tests "${CMAKE_CURRENT_SOURCE_DIR}/res/catalogue.json")
endif()
